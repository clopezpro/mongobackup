🚀 USO EN OTRA MÁQUINA
═══════════════════════════════════════════════════════════════════

1. Crear directorio:
   
   mkdir -p mongo-backup/{secrets,backups}
   cd mongo-backup

2. Crear docker-compose.yml:

   version: '3.7'
   secrets:
     backup_username:
       file: ./secrets/backup_username
     backup_password:
       file: ./secrets/backup_password
     backup_host:
       file: ./secrets/backup_host
     backup_databases:
       file: ./secrets/backup_databases
   
   services:
     backup-mongo:
       image: clopezpro/mongo-backup:latest
       container_name: mongodb-backup
       volumes:
         - ./backups:/backups
       environment:
         - TZ=America/Guayaquil
         - BACKUP_PORT=27017
         - BACKUP_HOST_FILE=/run/secrets/backup_host
         - BACKUP_USERNAME_FILE=/run/secrets/backup_username
         - BACKUP_PASSWORD_FILE=/run/secrets/backup_password
         - BACKUP_DATABASES_FILE=/run/secrets/backup_databases
         - CRON_TIME=0 3 * * *
         - MAX_BACKUPS=10
         - INIT_BACKUP=1
       secrets:
         - backup_username
         - backup_password
         - backup_host
         - backup_databases
       restart: unless-stopped

3. Crear secretos:

   echo "admin" > secrets/backup_username
   echo "password123" > secrets/backup_password
   echo "mongodb" > secrets/backup_host
   cat > secrets/backup_databases <<EOF
   database1
   database2
   EOF

4. Ejecutar:

   docker-compose up -d
   docker-compose logs -f
